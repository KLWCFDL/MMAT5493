import numpy as np

def calculate_matrix_power(matrix, power):
    """
    Raises a square matrix to an integer power.
    """
    # Convert input list to a NumPy array if it isn't already
    matrix_arr = np.array(matrix)
    
    # Compute the matrix power using numpy's built-in linear algebra tool
    return np.linalg.matrix_power(matrix_arr, power)

# ==========================================
# Example 1: 2x2 Matrix raised to the power of 3
# ==========================================
matrix_2x2 = [
    [0.126, 0.874],
    [0.663, 0.337]
]
power_2 = 20
result_2x2 = calculate_matrix_power(matrix_2x2, power_2)

print(f"2x2 Matrix raised to the power of {power_2}:")
print(result_2x2)
print("-" * 30)

# ==========================================
# Example 2: 3x3 Matrix raised to the power of 2
# ==========================================
matrix_3x3 = [
    [0.9, 0.075, 0.025],
    [0.15, 0.8, 0.05],
    [0.25, 0.25, 0.5]
]
power_3 = 20
result_3x3 = calculate_matrix_power(matrix_3x3, power_3)

print(f"3x3 Matrix raised to the power of {power_3}:")
print(result_3x3)
